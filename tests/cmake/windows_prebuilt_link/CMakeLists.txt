cmake_minimum_required(VERSION 3.18)
project(windows_prebuilt_link_regression LANGUAGES CXX)
if(NOT DEFINED LIBDPLYR_CMAKE_FILE)
    message(FATAL_ERROR "Pass -DLIBDPLYR_CMAKE_FILE=/absolute/path/CMakeLists.txt")
endif()

# Exercise the production link block without configuring Rust or downloading DuckDB.
file(READ "${LIBDPLYR_CMAKE_FILE}" source)
string(FIND "${source}" "    if(WIN32)\n        # Prefer the shared DuckDB target" start)
string(FIND "${source}" "    # Platform-specific test linking" end)
if(start LESS 0 OR end LESS start)
    message(FATAL_ERROR "Cannot locate the integration target's DuckDB link block")
endif()
math(EXPR length "${end} - ${start}")
string(SUBSTRING "${source}" ${start} ${length} link_block)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/production_link.cmake" "${link_block}")

# DuckDB's external interface: capabilities and the static target are declared later.
function(duckdb_extension_statically_link)
endfunction()
function(link_extension_libraries target linkage)
    target_link_libraries(${target} ${linkage} duckdb_loadable_extensions)
endfunction()
set(WIN32 TRUE)
set(PREBUILT_BINARY "duckdb_static.lib")
set(DUCKDB_LIBRARIES duckdb)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/main.cpp"
    "int engine(); int capability(); int main() { return engine() + capability(); }\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/engine.cpp" "int engine() { return 0; }\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/capability.cpp" "int capability() { return 0; }\n")
add_executable(duckdb_extension_integration_test "${CMAKE_CURRENT_BINARY_DIR}/main.cpp")
include("${CMAKE_CURRENT_BINARY_DIR}/production_link.cmake")
add_library(duckdb_static STATIC "${CMAKE_CURRENT_BINARY_DIR}/engine.cpp")
add_library(duckdb_loadable_extensions STATIC "${CMAKE_CURRENT_BINARY_DIR}/capability.cpp")
enable_testing()
add_test(NAME windows_prebuilt_link COMMAND duckdb_extension_integration_test)
